	function searchBox() {

		$("#headerNav_uc_widgetContainer_ctl00_ctl01_ctl00_SearchBox1_tbx_searchBox").autocomplete({
			source: function (request, response) {
				$.ajax({
					type: "POST",
					url: SITEBASEURL + "services/SearchService.asmx/GetAutoFillTitles",

					contentType: "application/json; charset=utf-8",
					data: "{ 'resultCount': '12', 'searchString': '" + request.term + "' }",
					dataType: "json",
					success: function (data) {

						response($.map(data.d.ListItems, function (item) {

							var maxCharLength = 16;
							if (item.Name.length >= maxCharLength) {
								item.Name = item.Name.substr(0, maxCharLength) + " ...";
							}


							return {
								label: item.Name,
								value: item.URL
							}
						}));

					}
				});
			},
			minLength: 2,
			appendTo: "body",
			delay: 75,
			select: function (event, ui) {
				var searchURL = 'https://www.bccenter.org/search.aspx?q=' + escape(ui.item.value);
				window.location = searchURL;

			},
			open: function (event, ui) {
				var resultsHover = $(this).autocomplete("widget");
				var tbx = $(this);

				resultsHover.css("width", tbx.css("width") + " !important");
			}
		});



	}

	searchBox();

});


</script>

                            closeClonedSearchPopup();

        triggers.forEach(trigger => {
            const parentLi = trigger.parentElement;

            trigger.setAttribute('aria-expanded', 'false');

            // Toggle aria state based on hover/focus
            const updateState = (e) => {
                const isVisible = parentLi.matches(':hover, :focus-within') && !parentLi.classList.contains('is-dismissed');
                trigger.setAttribute('aria-expanded', isVisible);

                // reset when the user leaves the menu item entirely
                if (e.type === 'mouseleave' || e.type === 'focusout') {
                    if (!parentLi.matches(':hover, :focus-within')) {
                        parentLi.classList.remove('is-dismissed');
                    }
                }
            };

            ['mouseenter', 'mouseleave', 'focusin', 'focusout'].forEach(e => {
                parentLi.addEventListener(e, updateState);
            });
        });

        // Handle ESC Key 
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                const activeMenu = document.querySelector('.nav .group:hover, .nav .group:focus-within');                
                if (activeMenu) {
                    activeMenu.classList.add('is-dismissed');                    
                    const trigger = activeMenu.querySelector('a');
                    if (trigger) trigger.setAttribute('aria-expanded', 'false');
                }
            }
        });
    })();

    function showItemDescription(item, data) {

        removeItemDescription();

        var itemLeft = parseInt(position.left) + 11;

        var itemDetailIframe = $('<iframe class="itemDetailIframe" frameborder="0" scrolling="no" src="' + detailIframePath + '"></iframe>');
        itemDetailContainer.append(itemDetailIframe);

        var gal = $('.galleryItem').length;
		if(gal>0){
			itemDetailContainer.css('top', itemBottom-4 + 'px');
		} else {
			itemDetailContainer.css('top', itemBottom+1 + 'px');
		}

		if(itemLeft < 100 && esf_view == 'CalendarView'){
		    itemDetailContainer.css('left',itemLeft + 'px');
		}

		$('#eventScheduleContent').append(itemDetailContainer);
        $('.calloutArrow').css('display', 'none');
        $(item).find('.calloutArrow').css('display', 'block');
		$('.calloutArrowGallery').css('display', 'none');
		$(item).find('.calloutArrowGallery').css('display', 'block');

        var getWindowWidth = $(window).width();
        itemDetailContainer.hover(
        function () {
            clearTimeout(removeItemDescriptionTimeout);
        }
        , function () {
            if(!isResponsive){
                removeItemDescriptionTimeout = setTimeout(function () { removeItemDescription(); });

    function loadDateList() {

        var eventNum = eventDays.length;
        if(eventNum>0) {$('#eventDateList').append('<li class=""><a href="" onclick="setDayFilter(0, event);return false;" role="option" aria-selected="false" tabindex="-1">All</a></li>');}
        for (var i = 0; i < eventNum; i++) {

            var dayName = eventDaysName[i]

            if (dayName.indexOf('year') == -1 && dayName.indexOf('month') == -1) {

                if (eventDays[i] == today) {
                    dayName = 'Today'
                }



                if ($.inArray(eventDays[i], eventInProgressDays) > -1) {
                    $('#eventDateList').append('<li class="InProgressEventDate"><a href="" onclick="setDayFilter(\'' + eventDays[i] + '\', event);return false;" role="option" aria-selected="false"  tabindex="-1">' + dayName + '</a></li>');
                }
                else {
                    $('#eventDateList').append('<li class=""><a href="" onclick="setDayFilter(\'' + eventDays[i] + '\', event);return false;" aria-selected="false" role="option" tabindex="-1">' + dayName + '</a></li>');
                }
            }
        }
    }

$('.filter-group input[type="checkbox"]:not(#filter-toggle)').not('#date-toggle').on('keypress click', function(e) {
  var parentEl = $(this).parent();
  var id = $(this).attr('id');
  var customFilterToggle = document.getElementById(id);
  var dropdownList = parentEl.find('.scroll-panel ul');
  if (e.which === 32 || e.type === 'click') {
    if (!customFilterToggle.checked) {
      dropdownList.find('li a').attr('tabindex', "-1");
      customFilterToggle.checked = false;
      customFilterToggle.focus();
    } else {
      dropdownList.find('li a').attr('tabindex', "0");
      dropdownList.find('li a').first().focus();
      dropdownList.find('a').keyup(function(e) {
        if (e.keyCode === 27 || e.which === 13) {
          dropdownList.find('li a').attr('tabindex', "-1");
          customFilterToggle.checked = false;
          customFilterToggle.focus();
        }
      });
      dropdownList.find('li a').on('keydown', function(e) {
          if (e.key === 'ArrowDown') {
              e.preventDefault();
              $(this).parent().next().find('a').focus();
          }
      });
      dropdownList.find('li a').on('keydown', function(e) {
          if (e.key === 'ArrowUp') {
              e.preventDefault();
              $(this).parent().prev().find('a').focus();
          }
      });  
      dropdownList.find('li a').first().on('keydown', function(e) {
        if (e.which === 9) {
          if (e.shiftKey) {
            e.preventDefault();
            parentEl.find('.input-icon').focus();
          }
        }
      });

      customFilterToggle.focus();

      dropdownList.find('li a').attr('tabindex', "0");
      dropdownList.find('li a').first().focus();

        if (e.keyCode === 27 || e.which === 13) {
          dropdownList.find('li a').attr('tabindex', "-1");
          customFilterToggle.checked = false;
          customFilterToggle.focus();
        }
      });
      dropdownList.find('li a').on('keydown', function(e) {
          if (e.key === 'ArrowDown') {
              e.preventDefault();
              $(this).parent().next().find('a').focus();
          }
      });
      dropdownList.find('li a').on('keydown', function(e) {
          if (e.key === 'ArrowUp') {

        if (e.which === 9) {

            parentEl.focus();
            dropdownList.find('li a').attr('tabindex', "-1");
            customFilterToggle.checked = false;
          }
        }
      });
      dropdownList.find('li a').last().on('keydown', function(e) {
        if (e.which === 9) {
          e.preventDefault(); 
          dropdownList.find('li a').first().focus();

      });

  }
});

$('.filter-group span.input-icon').on('keydown', function(e) {
  var parentEl = $(this).parent();
  var toggle = $(this).siblings('input[type="checkbox"]');
  var filterToggle = document.getElementById('filter-toggle');
  var dropdownList = parentEl.find('.scroll-panel ul');
  var id = toggle.attr('id');
  var toggleChecked = document.getElementById(id);
  if (e.which === 9) {
    if (e.shiftKey) {
      e.preventDefault();
      toggle.focus();
      filterToggle.checked = false;
      toggleChecked.checked = false;
      if (toggleChecked.checked) {
        if (dropdownList.is('#eventFilterList')) {
          dropdownList.find('li').attr('tabindex', "-1");
        } else {
          dropdownList.find('li a').attr('tabindex', "-1");
        }
      }
    } 
    if (toggleChecked.checked) {
      if (dropdownList.is('#eventFilterList')) {
        dropdownList.find('li').first().focus();
      } else {
        dropdownList.find('li a').first().focus();
      }
    }
  }
  if (e.which === 13) {
    $(this).click();
    dropdownList.find('li a').attr('tabindex', "-1");
    toggle.focus();
  }
});

function toggleOff(id) {
  updateArray();
  var target = id || 'none';
  for (var i = 0; i < checked.length; i++) {
    if (!checked[i].is("#" + target)) {
      checked[i].prop('checked', false);
      checked[i].attr('aria-expanded', 'false');
      checked[i].siblings('.scroll-panel').attr('aria-hidden', 'true');
      checked[i].siblings('.scroll-panel').find('a[tabindex="0"], li[tabindex="0"]').attr('tabindex','-1');
    }
  }
  checked = [];
}

  }

});

function eventSearchBox() {

    $("#tbx_eventSearch").autocomplete({
        source: function (request, response) {
            $.ajax({
                type: "POST",
                url: SITEBASEURL + "services/SearchService.asmx/GetEventAutoFillTitles",

                contentType: "application/json; charset=utf-8",
                data: "{ 'resultCount': '12', 'searchString': '" + escape(request.term) + "','isolatedSchedule':'false' }",
                dataType: "json",
                success: function (data) {

                    response($.map(data.d.ListItems, function (item) {

                        var maxCharLength = 40;
                        if (item.Name.length >= maxCharLength) {
                            item.Name = item.Name.substr(0, maxCharLength) + " ...";
                        }


                        return {
                            label: item.Name,
                            value: item.URL
                        }
                    }));
                }
            });
        },
        minLength: 2,
        appendTo: "body",
        delay: 75,
        select: function (event, ui) {
            clearPickFilters();
            var searchKeyword = ui.item.value;          
            searchInDescription = true;
            
            $.ajax({
                type: "POST",
                url: SITEBASEURL + "services/SearchService.asmx/GetSingleEventSearchResult",
                contentType: "application/json; charset=utf-8",
                data: "{'searchString': '" + searchKeyword + "', 'searchInDescription':" + searchInDescription + "}",
                dataType: "json",
                success: function (data) {
                    if(data.d == 0){
                        setKeywordsFilter(searchKeyword);
                    }
                    else{
                        window.location = SITEBASEURL + 'events_detail.aspx?id=' + data.d
                    }
                },
                error: function(data){
                    setKeywordsFilter(searchKeyword);
                }
            });
        },
        open: function (event, ui) {
            var resultsHover = $(this).autocomplete("widget");

            var tbx = $(this);

            resultsHover.css("width", tbx.css("width") + " !important");
            $('.ui-autocomplete-input').css('border-radius','4px 4px 0 0');
        },
        close: function (event, ui) {
            $('.ui-autocomplete-input').css('border-radius','4px');
        }
    });



}

function initiateModuleSlideshows(){
    winWidth = $(window).width();

        $('.swiperMobileInstructions').css('display','block');

        setTimeout(function () {
            mySwiper.update();
        },1)
    }
}

          <script>
            $(document).ready(function() {
                    var $loginToggle = $('.mobile-menu-login .mobile-menu-subheader-btn');
                    var $loginLinksContainer = $('.mobile-menu-login-links');
                    var $subheader = $('.mobile-menu-subheader');

                    // Ensure links are hidden from tab order if menu is closed
                    $loginLinksContainer.find('a').attr('tabindex', '-1');

                    function toggleLoginMenu() {
                        var isExpanded = $loginToggle.attr('aria-expanded') === 'true';
                        
                        $subheader.toggleClass('mobile-menu-login-showlinks');
                        
                        $loginToggle.attr('aria-expanded', !isExpanded);
                        
                        // Manage focusability of the links
                        if (!isExpanded) {
                              $loginLinksContainer.find('a').attr('tabindex', '0');
                        } else {
                              $loginLinksContainer.find('a').attr('tabindex', '-1');
                        }
                    }

                    // Handle interaction
                    $loginToggle.on('click keydown', function(e) {
                        if (e.type === 'keydown') {
                            if (e.key !== 'Enter' && e.key !== ' ') return;
                            e.preventDefault(); 
                        }
                        toggleLoginMenu();
                    });
            });
          </script>
